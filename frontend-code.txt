  readonly mobileMenuOpen = signal(false);

    this.langMenuOpen.update((open) => !open);
  }

  toggleMobileMenu(): void {
    this.mobileMenuOpen.update((open) => !open);
  }

    this.langMenuOpen.set(false);
    this.mobileMenuOpen.set(false);

  updatePageSeo(
    pageKey: "home" | "about" | "projects" | "skills" | "contact",
  ): void {
    const metadata = buildSeoMetadata(pageKey);
    this.applySeoMetadata(metadata);
  }

  applySeoMetadata(metadata: SeoMetadata): void {
    this.title.setTitle(metadata.title);

    this.meta.updateTag({ name: "description", content: metadata.description });
    if (metadata.keywords) {
      this.meta.updateTag({ name: "keywords", content: metadata.keywords });
    }

    this.meta.updateTag({ property: "og:type", content: "website" });
    this.meta.updateTag({ property: "og:title", content: metadata.title });
    this.meta.updateTag({
      property: "og:description",
      content: metadata.description,
    });

    this.meta.updateTag({
      name: "twitter:card",
      content: "summary_large_image",
    });
    this.meta.updateTag({ name: "twitter:title", content: metadata.title });
    this.meta.updateTag({
      name: "twitter:description",
      content: metadata.description,
    });

    if (metadata.canonicalPath) {
      this.updateCanonicalLink(metadata.canonicalPath);
    }
  }

  private updateCanonicalLink(path: string): void {
    const head = this.document?.head;
    if (!head) return;

    let linkEl = head.querySelector<HTMLLinkElement>('link[rel="canonical"]');
    if (!linkEl) {
      linkEl = this.document.createElement("link");
      linkEl.setAttribute("rel", "canonical");
      head.appendChild(linkEl);
    }

    const origin =
      this.document.location?.origin && this.document.location.origin !== "null"
        ? this.document.location.origin
        : this.BASE_URL;

    const normalizedPath = path.startsWith("/") ? path : `/${path}`;
    linkEl.setAttribute("href", `${origin}${normalizedPath}`);
  }

export function filterProjectsByCategory(
  projects: readonly ProjectItem[],
  category: ProjectCategory,
  searchQuery = "",
): ProjectItem[] {
  const normalizedQuery = searchQuery.trim().toLowerCase();
  return projects.filter((project) => {
    const matchesCategory = category === "All" || project.category === category;
    if (!matchesCategory) {
      return false;
    }
    if (!normalizedQuery) {
      return true;
    }
    const inTitle = project.title.toLowerCase().includes(normalizedQuery);
    const inDesc = project.shortDescription
      .toLowerCase()
      .includes(normalizedQuery);
    const inTags = project.tags.some((t) =>
      t.toLowerCase().includes(normalizedQuery),
    );
    return inTitle || inDesc || inTags;
  });
}

export function validateContactFormInput(payload: ContactFormPayload): {
  valid: boolean;
  errors: Partial<Record<keyof ContactFormPayload, string>>;
} {
  const errors: Partial<Record<keyof ContactFormPayload, string>> = {};
  const trimmedName = payload.name.trim();
  const trimmedEmail = payload.email.trim();
  const trimmedMessage = payload.message.trim();

  if (!trimmedName || trimmedName.length < 2) {
    errors.name = "Please enter your name (at least 2 characters).";
  }

  const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  if (!trimmedEmail || !emailPattern.test(trimmedEmail)) {
    errors.email = "Please enter a valid email address.";
  }

  if (!trimmedMessage || trimmedMessage.length < 10) {
    errors.message = "Please enter a message with at least 10 characters.";
  }

  return {
    valid: Object.keys(errors).length === 0,
    errors,
  };
}

import { animate, stagger } from "motion";

export const ANIMATION_TIMINGS = {
  microFeedbackSec: 0.18,
  pageTransitionSec: 0.42,
  scrollRevealSec: 0.52,
  staggerStepSec: 0.08,
  easingCurve: [0.16, 1, 0.3, 1] as [number, number, number, number],
};

function prefersReducedMotion(): boolean {
  if (
    typeof window === "undefined" ||
    typeof window.matchMedia !== "function"
  ) {
    return true;
  }
  return window.matchMedia("(prefers-reduced-motion: reduce)").matches;
}

function applyFallbackStyles(element: HTMLElement): void {
  element.style.opacity = "1";
  element.style.transform = "none";
}

export function animatePageEnter(container: HTMLElement | null): void {
  if (!container) return;

  if (prefersReducedMotion()) {
    applyFallbackStyles(container);
    return;
  }

  animate(
    container,
    {
      opacity: [0, 1],
      y: [12, 0],
    },
    {
      duration: ANIMATION_TIMINGS.pageTransitionSec,
      ease: ANIMATION_TIMINGS.easingCurve,
    },
  );
}

export function animateElementReveal(
  element: HTMLElement | null,
  delaySec = 0,
): void {
  if (!element) return;

  if (prefersReducedMotion()) {
    applyFallbackStyles(element);
    return;
  }

  animate(
    element,
    {
      opacity: [0, 1],
      y: [18, 0],
    },
    {
      duration: ANIMATION_TIMINGS.scrollRevealSec,
      delay: delaySec,
      ease: ANIMATION_TIMINGS.easingCurve,
    },
  );
}

export function animateStaggerChildren(
  parent: HTMLElement | null,
  childSelector = "[data-animate-item]",
): void {
  if (!parent) return;

  const items = parent.querySelectorAll<HTMLElement>(childSelector);
  if (!items.length) return;

  if (prefersReducedMotion()) {
    items.forEach((item) => applyFallbackStyles(item));
    return;
  }

  animate(
    items,
    {
      opacity: [0, 1],
      y: [16, 0],
    },
    {
      delay: stagger(ANIMATION_TIMINGS.staggerStepSec),
      duration: ANIMATION_TIMINGS.scrollRevealSec,
      ease: ANIMATION_TIMINGS.easingCurve,
    },
  );
}

import { isPlatformBrowser } from '@angular/common';
import {
  AfterViewInit,
  DestroyRef,
  Directive,
  ElementRef,
  PLATFORM_ID,
  inject,
  input,
} from '@angular/core';
import { animateElementReveal } from '../animations/animations';

@Directive({
  selector: '[appScrollReveal]',
})
export class ScrollRevealDirective implements AfterViewInit {
  readonly revealDelay = input<number>(0);

  private readonly el = inject<ElementRef<HTMLElement>>(ElementRef);
  private readonly platformId = inject(PLATFORM_ID);
  
  private readonly destroyRef = inject(DestroyRef);

  ngAfterViewInit(): void {
    if (!isPlatformBrowser(this.platformId)) {
      return;
    }

    const nativeEl = this.el.nativeElement;

    if (typeof IntersectionObserver === 'undefined') {
      animateElementReveal(nativeEl, this.revealDelay());
      return;
    }

    const observer = new IntersectionObserver(
      (entries) => {
        for (const entry of entries) {
          if (entry.isIntersecting) {
            animateElementReveal(nativeEl, this.revealDelay());
            observer.unobserve(nativeEl);
          }
        }
      },
      {
        threshold: 0.08,
        rootMargin: '0px 0px -24px 0px',
      }
    );

    observer.observe(nativeEl);

    this.destroyRef.onDestroy(() => {
      observer.disconnect();
    });
  }
}
